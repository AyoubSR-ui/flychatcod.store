### 3c — Add `runOrderExtractionFlowSilent` function

Add this new function right after the existing `runOrderExtractionFlow` function:

```typescript
/**
 * Silent extraction — performs DB operations (order create/cancel/update)
 * WITHOUT inserting any bot messages. The AI conversational reply already
 * handles all customer-facing communication. This prevents double messages (Bug A).
 */
async function runOrderExtractionFlowSilent(
  conv: Conversation,
  store: Store,
  products: Product[],
  history: Message[]
): Promise<void> {
  const intent = classifyDominantIntent(history);

  // Skip extraction for non-actionable intents
  if (intent === "product_inquiry" || intent === "thanks_closing" || intent === "other") {
    return;
  }

  let extraction: OrderExtractionResult | null = null;
  try {
    extraction = await extractOrderState(history, products, store);
  } catch (err) {
    console.error("[extraction] OpenAI call failed:", err);
    return;
  }

  if (!extraction) return;

  // --- Handle order creation (silent) ---
  if (
    intent === "order_confirm" &&
    extraction.canAutoCreate &&
    extraction.orderData
  ) {
    await handleAiOrderCreationSilent(conv, store, extraction.orderData);
    return;
  }

  // --- Handle cancellation (silent) ---
  if (
    (intent === "cancel" || conv.aiFlowState === "pending_cancel_choice") &&
    extraction.cancelIntent &&
    extraction.customerPhone
  ) {
    await handleAiCancellationSilent(conv, store, extraction.customerPhone);
    return;
  }
}
```

/**
 * Creates order in DB without inserting a bot confirmation message.
 * Tags order as created by AI agent.
 */
async function handleAiOrderCreationSilent(
  conv: Conversation,
  store: Store,
  orderData: ExtractedOrderData
): Promise<void> {
  try {
    const orderId = generateId("ord");
    await db.insert(orders).values({
      id: orderId,
      storeId: store.id,
      conversationId: conv.id,
      customerId: conv.customerId ?? null,
      status: "awaiting_confirmation",

      customerPhone: orderData.customerPhone,
      wilaya: orderData.wilaya,
      address: orderData.address ?? null,
      notes: "Created by AI agent", // ← AI agent label
      createdAt: new Date(),
      updatedAt: new Date(),
    });

    // Insert order items
    for (const item of orderData.items ?? []) {
      await db.insert(orderItems).values({
        id: generateId("oi"),
        orderId,
        productId: item.productId ?? null,
        productName: item.productName,

        quantity: item.quantity,
        variant: item.variant ?? null,

    // Update conversation flow state
    await db
      .update(conversations)
      .set({ aiFlowState: "order_created", updatedAt: new Date() })
      .where(eq(conversations.id, conv.id));

    console.log(`[AI] Order ${orderId} created silently for conv ${conv.id}`);
  } catch (err) {
    console.error("[AI] Silent order creation failed:", err);

### 3e — Add `handleAiCancellationSilent` function

Add this function after `handleAiCancellation`:

```typescript
/**
 * Cancels order in DB without inserting a bot confirmation message.
 */
async function handleAiCancellationSilent(
  conv: Conversation,
  store: Store,
  customerPhone: string
): Promise<void> {
  try {
    const cancellableStatuses = ["new", "awaiting_confirmation", "confirmed"];

    const [targetOrder] = await db
      .select()
      .from(orders)
      .where(
        and(
          eq(orders.storeId, store.id),
          eq(orders.customerPhone, customerPhone),
          inArray(orders.status, cancellableStatuses)
        )
      )
      .orderBy(desc(orders.createdAt))
      .limit(1);

    if (!targetOrder) {
      console.log(`[AI] No cancellable order found for phone ${customerPhone}`);
      return;
    }

    await db
      .update(orders)
      .set({ status: "cancelled", updatedAt: new Date() })
      .where(eq(orders.id, targetOrder.id));

    await db
      .update(conversations)
      .set({ aiFlowState: "order_cancelled", updatedAt: new Date() })
      .where(eq(conversations.id, conv.id));

    console.log(`[AI] Order ${targetOrder.id} cancelled silently for conv ${conv.id}`);
  } catch (err) {
    console.error("[AI] Silent cancellation failed:", err);
  }
}
```
